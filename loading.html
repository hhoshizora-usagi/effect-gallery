中身が届くまで、灰色の仮の形に光がキラッと流れる。2秒後に本物のカードに入れかわる。

使いどころSNSのタイムライン・ニュース一覧・商品リスト

単体で開く ↗ 
<article class="fx-skel" data-fx="skeleton" data-delay="2000" aria-busy="true">
  <div class="fx-skel__avatar"></div>
  <div class="fx-skel__body">
    <div class="fx-skel__line" style="width: 55%"></div>

      <h3>うさぎ店長</h3>

  </template>
</article>

// 見本では data-delay ミリ秒後に <template> の中身へ入れかえる
// 本番では fetch() でデータが届いたタイミングで入れかえる
(() => {
  document.querySelectorAll('[data-fx="skeleton"]').forEach((card) => {
    const tpl = card.querySelector('template');
    setTimeout(() => {
      card.replaceChildren(tpl.content.cloneNode(true));
      card.classList.add('is-loaded');
      card.setAttribute('aria-busy', 'false');
    }, +card.dataset.delay || 2000);
  });
})();

04

進み具合リング

Progress Ring

  • HTML
  • CSS
  • JS

丸いゲージが0%から100%までぐるっと塗られていき、真ん中の数字も一緒に増えていく。

使いどころアップロード中・ダウンロード中・目標の達成率

単体で開く ↗ 
<div class="fx-ring" data-fx="progress-ring" role="progressbar" aria-valuemin="0" aria-valuemax="100" aria-valuenow="0" aria-label="アップロードの進み具合">
  <svg viewBox="0 0 120 120" aria-hidden="true">
    <defs>
      <linearGradient id="fx-ring-grad" x1="0" y1="0" x2="1" y2="1">
        <stop offset="0" stop-color="#ff3d7f"/><stop offset=".5" stop-color="#ffd400"/><stop offset="1" stop-color="#7cff4f"/>
      </linearGradient>
    </defs>
    <circle class="fx-ring__bg" cx="60" cy="60" r="50"/>
    <circle class="fx-ring__bar" cx="60" cy="60" r="50"/>
  </svg>
  <b class="fx-ring__num">0%</b>
</div>
.fx-ring { position: relative; width: 200px; height: 200px; }
.fx-ring svg { width: 100%; height: 100%; transform: rotate(-90deg); }   /* 12時の位置からスタート */
.fx-ring circle { fill: none; stroke-width: 14; }
.fx-ring__bg { stroke: rgba(255, 255, 255, .12); }
.fx-ring__bar {
  stroke: url(#fx-ring-grad);
  stroke-linecap: round;
  stroke-dasharray: 314.16;                           /* 円周 = 2 × π × 50 */
  stroke-dashoffset: calc(314.16 * (1 - var(--p, 0)));
}
.fx-ring__num {
  position: absolute;
  inset: 0;
  display: grid;
  place-items: center;
  color: #fff;
  font-size: 40px;
  font-weight: normal;
  font-variant-numeric: tabular-nums;
}
.fx-ring.is-done .fx-ring__num { font-size: 30px; color: #7cff4f; animation: fx-ring-pop .5s cubic-bezier(.2, 1.8, .4, 1); }
@keyframes fx-ring-pop { 50% { transform: scale(1.3); } }
// 見本では 0 → 100% を3秒で進める。本番では進み具合（0〜1）を setProgress に渡す
(() => {
  document.querySelectorAll('[data-fx="progress-ring"]').forEach((ring) => {
    const num = ring.querySelector('.fx-ring__num');
    const setProgress = (p) => {
      ring.style.setProperty('--p', p);
      num.textContent = p >= 1 ? 'DONE!' : `${Math.round(p * 100)}%`;
      ring.setAttribute('aria-valuenow', Math.round(p * 100));
      ring.classList.toggle('is-done', p >= 1);
    };
    const start = performance.now(), dur = 3000;
    const tick = (now) => {
      const t = Math.min(1, (now - start) / dur);
      setProgress(t < 1 ? t * (2 - t) : 1);              // 最後ゆっくり
      if (t < 1) requestAnimationFrame(tick);
    };
    requestAnimationFrame(tick);
  });
})();
