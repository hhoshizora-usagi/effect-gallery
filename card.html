01

3Dチルト＋光沢

3D Tilt & Glare

  • HTML
  • CSS
  • JS

マウスの位置に合わせてカードが立体的に傾き、表面に光がすべるように反射する。

使いどころ商品カード・チケット・メンバーカード・作品一覧

単体で開く ↗ 
<div class="fx-tilt" data-fx="tilt">
  <div class="fx-tilt__inner">
    <p class="fx-tilt__label">VIP PASS</p>
    <h3 class="fx-tilt__title">EPIC<br>DAY</h3>
    <p class="fx-tilt__no">No. 0001</p>
  </div>
</div>
.fx-tilt {
  --rx: 0deg;
  --ry: 0deg;
  --gx: 50%;
  --gy: 50%;
  width: min(70vw, 250px);
  aspect-ratio: 3 / 4;
  perspective: 800px;
}
.fx-tilt__inner {
  position: relative;
  display: flex;
  flex-direction: column;
  justify-content: space-between;
  height: 100%;
  padding: 24px;
  overflow: hidden;
  border: 4px solid #111;
  border-radius: 22px;
  background: linear-gradient(135deg, #ff3d7f, #a259ff 55%, #00c2ff);
  color: #fff;
  transform: rotateX(var(--rx)) rotateY(var(--ry));
  transform-style: preserve-3d;
  transition: transform .15s ease-out;
  box-shadow: 0 30px 40px -20px rgba(0, 0, 0, .7);
}
.fx-tilt.is-leaving .fx-tilt__inner { transition: transform .6s cubic-bezier(.2, .9, .3, 1.3); }
/* 光の反射 */
.fx-tilt__inner::after {
  content: "";
  position: absolute;
  inset: 0;
  pointer-events: none;
  background: radial-gradient(circle at var(--gx) var(--gy), rgba(255, 255, 255, .55), transparent 55%);
  mix-blend-mode: overlay;
}
.fx-tilt__label { margin: 0; font-size: 14px; letter-spacing: .3em; }
.fx-tilt__title { margin: 0; font-size: 56px; line-height: .9; transform: translateZ(40px); text-shadow: 4px 4px 0 #111; }
.fx-tilt__no { margin: 0; font-family: ui-monospace, monospace; }
// カードの中のマウス位置（0〜1）から、傾きの角度と光の位置を決める
(() => {
  const max = 18;   // いちばん傾いたときの角度
  document.querySelectorAll('[data-fx="tilt"]').forEach((card) => {
    card.addEventListener('pointermove', (e) => {
      const r = card.getBoundingClientRect();
      const x = (e.clientX - r.left) / r.width;
      const y = (e.clientY - r.top) / r.height;
      card.classList.remove('is-leaving');
      card.style.setProperty('--ry', `${(x - .5) * max * 2}deg`);
      card.style.setProperty('--rx', `${(.5 - y) * max * 2}deg`);
      card.style.setProperty('--gx', `${x * 100}%`);
      card.style.setProperty('--gy', `${y * 100}%`);
    });
    card.addEventListener('pointerleave', () => {
      card.classList.add('is-leaving');
      card.style.setProperty('--rx', '0deg');
      card.style.setProperty('--ry', '0deg');
    });
  });
})();

  -webkit-mask-composite: xor;
          mask-composite: exclude;

  content: "";
  position: absolute;
  inset: 0;
  z-index: -1;
  border-radius: inherit;
  background: radial-gradient(260px circle at var(--mx) var(--my), rgba(255, 61, 127, .16), transparent 60%);

.fx-sb h3 { margin: 0 0 .3em; font-size: 22px; }
.fx-sb p  { margin: 0; font-family: system-ui, sans-serif; font-size: 14px; color: #c9c3e6; }
// グループ全体でマウスを見て、カードごとに「カードから見たマウスの位置」を入れる
(() => {
  document.querySelectorAll('[data-fx="spotlight-border"]').forEach((group) => {
    const cards = [...group.children];
    group.addEventListener('pointermove', (e) => {
      cards.forEach((card) => {
        const r = card.getBoundingClientRect();
        card.style.setProperty('--mx', `${e.clientX - r.left}px`);
        card.style.setProperty('--my', `${e.clientY - r.top}px`);
      });
    });
    group.addEventListener('pointerleave', () => {
      cards.forEach((card) => {
        card.style.setProperty('--mx', '-999px');
        card.style.setProperty('--my', '-999px');
      });
    });
  });
})();
