イベントまでカウントダウン

Event Countdown

  • HTML
  • CSS
  • JS

イベント当日まで「あと何日・何時間・何分・何秒」かを表示。数字が変わるたびに、ぴょこっとはねる。

使いどころイベント告知・セール開始・発売日・新年のカウントダウン

単体で開く ↗ 
<div class="fx-cd" data-fx="countdown" data-target="2026-11-23T10:00:00+09:00">
  <p class="fx-cd__title">EPIC DAY まで あと</p>
  <div class="fx-cd__row" role="timer" aria-live="off">
    <div class="fx-cd__box"><b data-unit="d">00</b><small>日</small></div>
    <div class="fx-cd__box"><b data-unit="h">00</b><small>時間</small></div>
    <div class="fx-cd__box"><b data-unit="m">00</b><small>分</small></div>
    <div class="fx-cd__box"><b data-unit="s">00</b><small>秒</small></div>
  </div>
</div>

// data-target の日時との差を、毎秒「日・時・分・秒」に分けて表示する
(() => {
  const pad = (n) => String(n).padStart(2, '0');
  document.querySelectorAll('[data-fx="countdown"]').forEach((root) => {
    let target = new Date(root.dataset.target).getTime();
    // 見本用：日付を過ぎていたら、7日後を目標にしてずっと動くようにする
    if (!(target > Date.now())) target = Date.now() + 7 * 864e5;
    const els = Object.fromEntries([...root.querySelectorAll('[data-unit]')].map((b) => [b.dataset.unit, b]));
    const set = (unit, val) => {
      const el = els[unit];
      if (el.textContent === val) return;
      el.textContent = val;
      el.classList.remove('is-tick');
      void el.offsetWidth;
      el.classList.add('is-tick');
    };
    const tick = () => {
      const left = Math.max(0, target - Date.now());
      const s = Math.floor(left / 1000);
      set('d', pad(Math.floor(s / 86400)));
      set('h', pad(Math.floor(s / 3600) % 24));
      set('m', pad(Math.floor(s / 60) % 60));
      set('s', pad(s % 60));
      root.querySelector('[role="timer"]').setAttribute('aria-label',
        `あと ${Math.floor(s / 86400)}日 ${Math.floor(s / 3600) % 24}時間 ${Math.floor(s / 60) % 60}分`);
    };
    tick();
    setInterval(tick, 1000);
  });
})();

// 1秒ごとに時刻を6桁にして、変わった桁だけ「古い数字の上半分」をめくって見せる
(() => {
  document.querySelectorAll('[data-fx="flip-clock"]').forEach((clock) => {
    const cards = [...clock.querySelectorAll('.fx-flipc__card')];
    const now = () => {
      const d = new Date();
      return [d.getHours(), d.getMinutes(), d.getSeconds()].map((n) => String(n).padStart(2, '0')).join('');
    };
    const update = () => {
      const digits = now();
      digits.split('').forEach((ch, i) => {
        const card = cards[i];
        const old = card.dataset.v;
        if (old === ch) return;
        card.dataset.v = ch;
        if (old !== undefined) {
          const leaf = document.createElement('span');
          leaf.className = 'fx-flipc__leaf';
          leaf.innerHTML = `<span>${old}</span>`;
          card.appendChild(leaf);
          leaf.addEventListener('animationend', () => leaf.remove());
        }
        card.firstChild && card.firstChild.nodeType === 3 ? (card.firstChild.nodeValue = ch) : card.prepend(ch);
      });
      clock.setAttribute('aria-label', `現在の時刻 ${digits.slice(0, 2)}時${digits.slice(2, 4)}分`);
    };
    update();
    setInterval(update, 1000);
  });
})();

// performance.now() で経過時間を計る（setInterval の数え上げよりズレにくい）
(() => {
  const fmt = (ms) => {
    const m = Math.floor(ms / 60000), s = Math.floor(ms / 1000) % 60, cs = Math.floor(ms / 10) % 100;
    return `${String(m).padStart(2, '0')}:${String(s).padStart(2, '0')}.${String(cs).padStart(2, '0')}`;
  };
  document.querySelectorAll('[data-fx="stopwatch"]').forEach((root) => {
    const time = root.querySelector('.fx-sw__time');
    const startBtn = root.querySelector('.fx-sw__start');
    const lapBtn = root.querySelector('.fx-sw__lap');
    const laps = root.querySelector('.fx-sw__laps');
    let base = 0, startAt = 0, running = false, raf;
    const elapsed = () => base + (running ? performance.now() - startAt : 0);
    const draw = () => {
      const ms = elapsed();
      time.textContent = fmt(ms);
      root.style.setProperty('--p', ((ms % 1000) / 10).toFixed(1));   // 1秒で1周
      if (running) raf = requestAnimationFrame(draw);
    };
    startBtn.addEventListener('click', () => {
      if (running) { base = elapsed(); running = false; cancelAnimationFrame(raf); }
      else { startAt = performance.now(); running = true; draw(); }
      startBtn.textContent = running ? 'ストップ' : 'スタート';
      startBtn.classList.toggle('is-running', running);
      lapBtn.disabled = !running;
    });
    lapBtn.addEventListener('click', () => {
      const li = document.createElement('li');
      li.textContent = `${laps.children.length + 1}: ${fmt(elapsed())}`;
      laps.appendChild(li);
      if (laps.children.length > 6) laps.firstElementChild.remove();
    });
    root.querySelector('.fx-sw__reset').addEventListener('click', () => {
      base = 0; startAt = performance.now();
      laps.innerHTML = '';
      if (!running) draw();
    });
    draw();
  });
})();
