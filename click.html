01

紙吹雪クラッカー

Confetti Burst

  • HTML
  • CSS
  • JS

クリックした場所から、紙吹雪がパーンとはじけて重力で落ちていく。ボタンを押すと特大。

使いどころ購入完了・応募ボタン・ゲームのクリア演出

単体で開く ↗ 
<canvas class="fx-burstcv" data-fx="confetti-burst" aria-hidden="true"></canvas>
<button class="fx-burst-btn" data-burst="big">🎉 おめでとう！</button>
.fx-burstcv {
  position: fixed;
  inset: 0;
  width: 100%;
  height: 100%;
  pointer-events: none;
  z-index: 9999;
}
.fx-burst-btn {
  padding: 1em 1.8em;
  border: 3px solid #111;
  border-radius: 14px;
  background: #7cff4f;
  color: #111;
  font: inherit;
  font-size: 22px;
  cursor: pointer;
  box-shadow: 6px 6px 0 #111;
}
.fx-burst-btn:active { transform: translate(3px, 3px); box-shadow: 3px 3px 0 #111; }
// クリックした点から、ランダムな方向・速さで紙を飛ばす（重力と空気抵抗つき）
(() => {
  const colors = ['#ff3d7f', '#ffd400', '#00c2ff', '#7cff4f', '#ff7a00', '#a259ff'];
  const cv = document.querySelector('[data-fx="confetti-burst"]');
  if (!cv) return;
  const ctx = cv.getContext('2d');
  const bits = [];
  const resize = () => {
    const d = window.devicePixelRatio || 1;
    cv.width = innerWidth * d; cv.height = innerHeight * d;
    ctx.setTransform(d, 0, 0, d, 0, 0);
  };
  resize();
  addEventListener('resize', resize);

  const burst = (x, y, n) => {
    for (let i = 0; i < n; i++) {
      const a = Math.random() * Math.PI * 2;
      const s = 4 + Math.random() * (n > 60 ? 12 : 7);
      bits.push({
        x, y, vx: Math.cos(a) * s, vy: Math.sin(a) * s - 4,
        w: 5 + Math.random() * 6, h: 8 + Math.random() * 8,
        rot: Math.random() * 6, vr: (Math.random() - .5) * .4,
        life: 1, c: colors[i % colors.length],
      });
    }
  };
  addEventListener('pointerdown', (e) => {
    const big = e.target.closest('[data-burst="big"]');
    burst(e.clientX, e.clientY, big ? 120 : 40);
  });

  (function loop() {
    ctx.clearRect(0, 0, innerWidth, innerHeight);
    for (let i = bits.length - 1; i >= 0; i--) {
      const b = bits[i];
      b.vy += .25; b.vx *= .985; b.vy *= .985;   // 重力と空気抵抗
      b.x += b.vx; b.y += b.vy; b.rot += b.vr; b.life -= .008;
      if (b.life <= 0 || b.y > innerHeight + 40) { bits.splice(i, 1); continue; }
      ctx.save();
      ctx.globalAlpha = Math.min(1, b.life * 2);
      ctx.translate(b.x, b.y); ctx.rotate(b.rot);
      ctx.fillStyle = b.c;
      ctx.fillRect(-b.w / 2, -b.h / 2, b.w, b.h);
      ctx.restore();
    }
    requestAnimationFrame(loop);
  })();
})();

02

波紋ボタン

Ripple

  • HTML
  • CSS
  • JS

押した場所から丸い波紋が広がるボタン。どこを押したかが気持ちよく伝わる。

使いどころフォームの送信・メニュー・あらゆるボタン

単体で開く ↗ 
<div class="fx-ripple-row">
  <button class="fx-ripple" data-fx="ripple">SUBMIT</button>
  <button class="fx-ripple fx-ripple--cyan" data-fx="ripple">MENU</button>
  <button class="fx-ripple fx-ripple--yellow" data-fx="ripple">LIKE</button>
</div>
.fx-ripple-row { display: flex; flex-wrap: wrap; gap: 20px; justify-content: center; }
.fx-ripple {
  position: relative;
  overflow: hidden;          /* 波紋をボタンの中だけに */
  padding: 1em 2.2em;
  border: 0;
  border-radius: 12px;
  background: #ff3d7f;
  color: #fff;
  font: inherit;
  font-size: 20px;
  cursor: pointer;
}
.fx-ripple--cyan   { background: #00c2ff; color: #111; }
.fx-ripple--yellow { background: #ffd400; color: #111; }
.fx-ripple__wave {
  position: absolute;
  width: 20px;
  height: 20px;
  margin: -10px 0 0 -10px;
  border-radius: 50%;
  background: rgba(255, 255, 255, .7);
  pointer-events: none;
  animation: fx-ripple .7s ease-out forwards;
}
@keyframes fx-ripple {
  to { transform: scale(var(--scale, 20)); opacity: 0; }
}
// 押した位置に丸を置いて、ボタン全体をおおう大きさまで広げてから消す
(() => {
  document.querySelectorAll('[data-fx="ripple"]').forEach((btn) => {
    btn.addEventListener('pointerdown', (e) => {
      const r = btn.getBoundingClientRect();
      const wave = document.createElement('span');
      wave.className = 'fx-ripple__wave';
      wave.style.left = `${e.clientX - r.left}px`;
      wave.style.top = `${e.clientY - r.top}px`;
      wave.style.setProperty('--scale', (Math.hypot(r.width, r.height) * 2) / 20);
      btn.appendChild(wave);
      wave.addEventListener('animationend', () => wave.remove());
    });
  });
})();

// クリックのたびにクラスをつけ直して、アニメーションを最初から再生する
(() => {
  document.querySelectorAll('[data-fx="jelly"]').forEach((btn) => {
    btn.addEventListener('click', () => {
      btn.classList.remove('is-boing');
      void btn.offsetWidth;            // つけ直すための小ワザ（再描画させる）
      btn.classList.add('is-boing');
    });
    btn.addEventListener('animationend', () => btn.classList.remove('is-boing'));
  });
})();

押すとボタンのまわりに星がキラッと飛び散る。何度でも押したくなる。

使いどころお気に入り・保存・「いいね」・ポイント獲得

単体で開く ↗ 
<button class="fx-sparkle" data-fx="sparkle">★ FAVORITE</button>
.fx-sparkle {

  position: absolute;

  background: var(--c);

  clip-path: polygon(50% 0, 61% 35%, 98% 35%, 68% 57%, 79% 91%, 50% 70%, 21% 91%, 32% 57%, 2% 35%, 39% 35%);

@keyframes fx-sparkle {
  0%   { transform: translate(0, 0) scale(0) rotate(0); opacity: 1; }
  60%  { opacity: 1; }
  100% { transform: translate(var(--x), var(--y)) scale(1) rotate(var(--r)); opacity: 0; }
}
// 星を14個作って、円形にばらけた方向へ飛ばす（飛んだら消す）
(() => {
  const colors = ['#ffd400', '#ff3d7f', '#7cff4f', '#ffffff', '#ff7a00', '#a259ff'];
  document.querySelectorAll('[data-fx="sparkle"]').forEach((btn) => {
    btn.addEventListener('click', () => {
      const n = 14;
      for (let i = 0; i < n; i++) {
        const star = document.createElement('i');
        star.className = 'fx-sparkle__star';
        const a = (i / n) * Math.PI * 2 + Math.random() * .4;
        const dist = 70 + Math.random() * 70;
        star.style.setProperty('--x', `${Math.cos(a) * dist}px`);
        star.style.setProperty('--y', `${Math.sin(a) * dist}px`);
        star.style.setProperty('--r', `${Math.random() * 360}deg`);
        star.style.setProperty('--s', `${12 + Math.random() * 16}px`);
        star.style.setProperty('--c', colors[i % colors.length]);
        btn.appendChild(star);
        star.addEventListener('animationend', () => star.remove());
      }
    });
  });
})();

// 押すたびに オン/オフ を切りかえて、数字を ±1。オンのときだけ丸を飛ばす
(() => {
  const colors = ['#ff3d7f', '#ffd400', '#00c2ff', '#7cff4f', '#a259ff'];
  document.querySelectorAll('[data-fx="like"]').forEach((btn) => {
    const count = btn.querySelector('.fx-like__count');
    const heart = btn.querySelector('.fx-like__heart');
    btn.addEventListener('click', () => {
      const on = btn.getAttribute('aria-pressed') !== 'true';
      btn.setAttribute('aria-pressed', String(on));
      count.textContent = +count.textContent + (on ? 1 : -1);
      if (!on) return;
      for (let i = 0; i < 10; i++) {
        const dot = document.createElement('i');
        dot.className = 'fx-like__dot';
        const a = (i / 10) * Math.PI * 2;
        dot.style.setProperty('--x', `${Math.cos(a) * 42}px`);
        dot.style.setProperty('--y', `${Math.sin(a) * 42}px`);
        dot.style.setProperty('--c', colors[i % colors.length]);
        heart.appendChild(dot);
        dot.addEventListener('animationend', () => dot.remove());
      }
    });
  });
})();

  100% { transform: scale(8); opacity: 0; border-width: 1px; }
}
// クリック位置に、少しずつ遅らせた輪を3つ置く
(() => {
  const set = ['#ff3d7f', '#ffd400', '#00c2ff'];
  document.querySelectorAll('[data-fx="shockwave"]').forEach((area) => {
    area.addEventListener('pointerdown', (e) => {
      const r = area.getBoundingClientRect();
      set.forEach((c, i) => {
        const ring = document.createElement('i');
        ring.className = 'fx-shock__ring';
        ring.style.left = `${e.clientX - r.left}px`;
        ring.style.top = `${e.clientY - r.top}px`;
        ring.style.setProperty('--c', c);
        ring.style.setProperty('--delay', `${i * .08}s`);
        area.appendChild(ring);
        ring.addEventListener('animationend', () => ring.remove());
      });
    });
  });
})();
