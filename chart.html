// 画面に入ったら .is-in をつけて棒を伸ばし、数字も 0 → --v まで数える
(() => {
  // 画面に入ったら1回だけ動かす
  const onVisible = (el, fn) => {
    const io = new IntersectionObserver((entries) => {
      if (entries[0].isIntersecting) { fn(); io.disconnect(); }
    }, { threshold: .4 });
    io.observe(el);
  };

  document.querySelectorAll('[data-fx="bar-grow"]').forEach((chart) => {
    const cols = [...chart.querySelectorAll('.fx-bars__col')];
    cols.forEach((c, i) => c.style.setProperty('--i', i));
    onVisible(chart, () => {
      chart.classList.add('is-in');
      cols.forEach((c, i) => {
        const to = parseFloat(getComputedStyle(c).getPropertyValue('--v'));
        const b = c.querySelector('b');
        const start = performance.now() + i * 120;
        const tick = (now) => {
          const t = Math.min(1, Math.max(0, (now - start) / 1200));
          b.textContent = Math.round(to * (1 - Math.pow(1 - t, 3))) + '%';
          if (t < 1) requestAnimationFrame(tick);
        };
        requestAnimationFrame(tick);
      });
    });
  });
})();

// 画面に入ったら、1つずつ時間差で円を描く。凡例とグラフのホバーを連動させる
(() => {
  // 画面に入ったら1回だけ動かす
  const onVisible = (el, fn) => {
    const io = new IntersectionObserver((entries) => {
      if (entries[0].isIntersecting) { fn(); io.disconnect(); }
    }, { threshold: .4 });
    io.observe(el);
  };

  document.querySelectorAll('[data-fx="donut"]').forEach((chart) => {
    const segs = [...chart.querySelectorAll('.fx-donut__seg')];
    const items = [...chart.querySelectorAll('.fx-donut__legend li')];
    segs.forEach((s, i) => s.style.setProperty('--i', i));
    onVisible(chart, () => chart.classList.add('is-in'));
    const hot = (n) => {
      segs.forEach((s, i) => s.classList.toggle('is-hot', i === n));
      items.forEach((li, i) => li.classList.toggle('is-hot', i === n));
    };
    [...segs, ...items].forEach((el) => {
      const n = segs.includes(el) ? segs.indexOf(el) : items.indexOf(el);
      el.addEventListener('mouseenter', () => hot(n));
      el.addEventListener('mouseleave', () => hot(-1));
    });
  });
})();

// ばねの動き（行きすぎて戻る）で --p を 0 → 目標値まで動かす。針とゲージは --p で連動
(() => {
  // 画面に入ったら1回だけ動かす
  const onVisible = (el, fn) => {
    const io = new IntersectionObserver((entries) => {
      if (entries[0].isIntersecting) { fn(); io.disconnect(); }
    }, { threshold: .4 });
    io.observe(el);
  };

  document.querySelectorAll('[data-fx="gauge"]').forEach((g) => {
    const target = Math.min(100, +g.dataset.value || 0);
    const num = g.querySelector('.fx-gauge__num');
    onVisible(g, () => {
      let p = 0, v = 0;
      (function step() {
        v += (target - p) * .02;        // 目標へ引っぱる力
        v *= .88;                       // 空気抵抗
        p += v;
        g.style.setProperty('--p', Math.max(0, Math.min(100, p)).toFixed(2));
        num.textContent = Math.round(Math.max(0, p));
        if (Math.abs(v) > .01 || Math.abs(target - p) > .1) requestAnimationFrame(step);
        else { g.style.setProperty('--p', target); num.textContent = target; }
      })();
    });
  });
})();

// data-values の数字から線・面・点を作り、画面に入ったら左から描く
(() => {
  // 画面に入ったら1回だけ動かす
  const onVisible = (el, fn) => {
    const io = new IntersectionObserver((entries) => {
      if (entries[0].isIntersecting) { fn(); io.disconnect(); }
    }, { threshold: .4 });
    io.observe(el);
  };

  const NS = 'http://www.w3.org/2000/svg';
  document.querySelectorAll('[data-fx="line-draw"]').forEach((fig) => {
    const vals = fig.dataset.values.split(',').map(Number);
    const max = Math.max(...vals) * 1.1, W = 400, H = 200;
    const pts = vals.map((v, i) => [(i / (vals.length - 1)) * W, H - (v / max) * H]);
    const d = pts.map(([x, y], i) => `${i ? 'L' : 'M'}${x.toFixed(1)} ${y.toFixed(1)}`).join(' ');
    const path = fig.querySelector('.fx-line__path');
    path.setAttribute('d', d);
    fig.querySelector('.fx-line__area').setAttribute('d', `${d} L${W} ${H} L0 ${H} Z`);
    path.style.setProperty('--len', path.getTotalLength() * 2);   // 線の長さぶん隠しておく
    const dots = fig.querySelector('.fx-line__dots');
    pts.forEach(([x, y], i) => {
      const c = document.createElementNS(NS, 'circle');
      c.setAttribute('cx', x); c.setAttribute('cy', y); c.setAttribute('r', 6);
      c.style.setProperty('--d', `${.2 + (i / (pts.length - 1)) * 1.4}s`);
      dots.appendChild(c);
    });
    onVisible(fig, () => fig.classList.add('is-in'));
  });
})();
