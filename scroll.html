スクロール進み具合バー

Progress Bar

  • HTML
  • CSS
  • JS

画面のいちばん上に虹色のバー。ページをどこまで読んだか、スクロールに合わせて伸びる。

使いどころ長い記事・ブログ・ランディングページ

単体で開く ↗ 
<div class="fx-progress" data-fx="progress" aria-hidden="true"></div>

<!-- ↓ スクロールを試すためのダミーの中身 -->
<section class="demo-block">SCROLL ↓</section>
<section class="demo-block demo-block--pink">LINEUP</section>
<section class="demo-block demo-block--yellow">TIMETABLE</section>
<section class="demo-block demo-block--cyan">TICKET</section>
<section class="demo-block">THE END</section>
.fx-progress {
  position: fixed;
  top: 0;
  left: 0;
  z-index: 9999;
  width: 100%;
  height: 8px;
  background: linear-gradient(90deg, #ff3d7f, #ff7a00, #ffd400, #7cff4f, #00c2ff, #a259ff);
  transform: scaleX(var(--p, 0));
  transform-origin: left;
  box-shadow: 0 2px 0 #111;
}
// スクロールした割合（0〜1）を CSS 変数 --p に入れる
(() => {
  const bars = document.querySelectorAll('[data-fx="progress"]');
  const update = () => {
    const max = document.documentElement.scrollHeight - innerHeight;
    const p = max > 0 ? scrollY / max : 0;
    bars.forEach((b) => b.style.setProperty('--p', p.toFixed(4)));
  };
  addEventListener('scroll', update, { passive: true });
  addEventListener('resize', update);
  update();
})();

})();

// data-speed の値 × スクロール量 だけ、上下にずらす（マイナスなら逆向き）
(() => {
  const els = [...document.querySelectorAll('[data-fx="parallax"]')];
  let ticking = false;
  const update = () => {
    els.forEach((el) => {
      const speed = parseFloat(el.dataset.speed) || 0;
      el.style.transform = `translate3d(0, ${scrollY * speed}px, 0)`;
    });
    ticking = false;
  };
  addEventListener('scroll', () => {
    if (!ticking) { requestAnimationFrame(update); ticking = true; }
  }, { passive: true });
  update();
})();

// 帯が画面の中を通るあいだ、スクロール量に合わせて横へずらす
(() => {
  const rows = [...document.querySelectorAll('[data-fx="scroll-marquee"]')];
  const update = () => {
    rows.forEach((row) => {
      const r = row.getBoundingClientRect();
      const progress = innerHeight - r.top;          // 画面に入ってから進んだ量
      const speed = parseFloat(row.dataset.speed) || .5;
      const start = speed > 0 ? 0 : -row.scrollWidth * .3;   // 右へ流す帯は左に余白を持たせる
      row.style.transform = `translate3d(${start + progress * -speed}px, 0, 0)`;
    });
  };
  addEventListener('scroll', () => requestAnimationFrame(update), { passive: true });
  addEventListener('resize', update);
  update();
})();

// 1フレームあたりのスクロール量（速さ）を傾きの角度に変える
(() => {
  const els = document.querySelectorAll('[data-fx="velocity-skew"]');
  let lastY = scrollY, skew = 0;
  (function loop() {
    const v = scrollY - lastY;            // スクロールの速さ
    lastY = scrollY;
    const target = Math.max(-12, Math.min(12, v * .6));
    skew += (target - skew) * .12;        // なめらかに追いかける
    els.forEach((el) => el.style.setProperty('--skew', `${skew.toFixed(2)}deg`));
    requestAnimationFrame(loop);
  })();
})();

// 画面に入ったら、1.8秒かけて 0 → data-to まで数字を増やす
(() => {
  const ease = (t) => 1 - Math.pow(1 - t, 3);      // 最後にゆっくり止まる
  const run = (el) => {
    const to = parseFloat(el.dataset.to) || 0;
    const dec = parseInt(el.dataset.decimals || '0', 10);
    const suffix = el.dataset.suffix || '';
    const dur = 1800, start = performance.now();
    const tick = (now) => {
      const t = Math.min(1, (now - start) / dur);
      const val = to * ease(t);
      el.textContent = val.toLocaleString('ja-JP', { minimumFractionDigits: dec, maximumFractionDigits: dec }) + suffix;
      if (t < 1) requestAnimationFrame(tick);
    };
    requestAnimationFrame(tick);
  };
  const io = new IntersectionObserver((entries) => {
    entries.forEach((en) => {
      if (en.isIntersecting) { run(en.target); io.unobserve(en.target); }
    });
  }, { threshold: .6 });
  document.querySelectorAll('[data-fx="counter"]').forEach((el) => io.observe(el));
})();

単体で開く ↗ 
<div class="fx-hue" data-fx="bg-color-shift" aria-hidden="true"></div>
<main class="fx-hue-content">
  <section class="demo-block">SCROLL ↓</section>
  <section class="demo-block">SECTION 02</section>
  <section class="demo-block">SECTION 03</section>
  <section class="demo-block">SECTION 04</section>
  <section class="demo-block">THE END</section>
</main>
/* 背景いっぱいに敷く1枚。スクロール量に合わせて色相（--hue）が変わる */
.fx-hue {
  position: fixed;
  inset: 0;
  z-index: 0;
  background: hsl(var(--hue, 260) 72% 22%);
}
.fx-hue-content { position: relative; z-index: 1; }
// スクロールの進み具合を0〜1で取り、色相（hue）を移り変わらせる
(() => {
  document.querySelectorAll('[data-fx="bg-color-shift"]').forEach(() => {
    const update = () => {
      const max = Math.max(1, document.documentElement.scrollHeight - innerHeight);
      const p = Math.min(1, Math.max(0, scrollY / max));
      document.documentElement.style.setProperty('--hue', (260 + p * 220).toFixed(1));
    };
    addEventListener('scroll', update, { passive: true });
    addEventListener('resize', update);
    update();
  });
})();
