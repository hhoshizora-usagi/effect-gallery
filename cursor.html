01

カラフル軌跡

Color Trail

  • HTML
  • CSS
  • JS

マウスやタッチの通ったあとに、カラフルな粒がふわっと残って消えていく。

使いどころトップページ全体・遊び心を出したいページ

単体で開く ↗ 
<canvas class="fx-trail" data-fx="trail" aria-hidden="true"></canvas>
.fx-trail {
  position: fixed;
  inset: 0;
  width: 100%;
  height: 100%;
  pointer-events: none;   /* 下のボタンなどを押せるようにする */
  z-index: 9999;
}
// マウスの通り道に粒を置いて、少しずつ小さく・薄くする
(() => {
  const colors = ['#ff3d7f', '#ffd400', '#00c2ff', '#7cff4f', '#ff7a00', '#a259ff'];
  document.querySelectorAll('[data-fx="trail"]').forEach((cv) => {
    const ctx = cv.getContext('2d');
    const dots = [];
    let last = null, n = 0;

    const resize = () => {
      const d = window.devicePixelRatio || 1;
      cv.width = innerWidth * d; cv.height = innerHeight * d;
      ctx.setTransform(d, 0, 0, d, 0, 0);
    };
    resize();
    addEventListener('resize', resize);

    addEventListener('pointermove', (e) => {
      const x = e.clientX, y = e.clientY;
      // 速く動かしても途切れないように、間を埋める
      const steps = last ? Math.min(8, Math.ceil(Math.hypot(x - last.x, y - last.y) / 10)) : 1;
      for (let i = 1; i <= steps; i++) {
        const t = i / steps;
        dots.push({
          x: last ? last.x + (x - last.x) * t : x,
          y: last ? last.y + (y - last.y) * t : y,
          r: 6 + Math.random() * 9,
          vx: (Math.random() - .5) * 1.2,
          vy: (Math.random() - .5) * 1.2 + .4,
          life: 1,
          c: colors[n++ % colors.length],
        });
      }
      last = { x, y };
    });
    document.addEventListener('pointerleave', () => { last = null; });

    (function loop() {
      ctx.clearRect(0, 0, innerWidth, innerHeight);
      for (let i = dots.length - 1; i >= 0; i--) {
        const p = dots[i];
        p.life -= .022; p.x += p.vx; p.y += p.vy; p.r *= .975;
        if (p.life <= 0) { dots.splice(i, 1); continue; }
        ctx.globalAlpha = p.life;
        ctx.fillStyle = p.c;
        ctx.beginPath(); ctx.arc(p.x, p.y, p.r, 0, Math.PI * 2); ctx.fill();
      }
      ctx.globalAlpha = 1;
      requestAnimationFrame(loop);
    })();
  });
})();

// ボタンの中心とマウスの距離を測って、近ければマウスの方へずらす
(() => {
  const els = [...document.querySelectorAll('[data-fx="magnetic"]')];
  const range = 50;       // ボタンのまわり何pxまで反応するか
  const power = .35;      // 吸い寄せの強さ
  els.forEach((el) => { el._tx = 0; el._ty = 0; });

  addEventListener('pointermove', (e) => {
    els.forEach((el) => {
      const r = el.getBoundingClientRect();
      // いまずれている分を引いて、元の中心を求める
      const cx = r.left + r.width / 2 - el._tx;
      const cy = r.top + r.height / 2 - el._ty;
      const dx = e.clientX - cx, dy = e.clientY - cy;
      const reach = range + Math.max(r.width, r.height) / 2;
      const near = Math.hypot(dx, dy) < reach;
      el._tx = near ? dx * power : 0;
      el._ty = near ? dy * power : 0;
      el.style.transform = `translate(${el._tx}px, ${el._ty}px)`;
      el.firstElementChild.style.transform = `translate(${el._tx * .4}px, ${el._ty * .4}px)`;
      el.classList.toggle('is-near', near);
    });
  });
})();

// 点はマウスにぴったり、輪っかは少し遅れてなめらかについてくる
(() => {
  if (!matchMedia('(pointer: fine)').matches) return;   // スマホでは使わない
  document.querySelectorAll('[data-fx="ring-cursor"]').forEach((root) => {
    const dot = root.querySelector('.fx-cursor__dot');
    const ring = root.querySelector('.fx-cursor__ring');
    let mx = -100, my = -100, rx = mx, ry = my;
    document.documentElement.classList.add('fx-cursor-on');

    addEventListener('pointermove', (e) => {
      mx = e.clientX; my = e.clientY;
      dot.style.transform = `translate(${mx}px, ${my}px)`;
      root.classList.add('is-visible');
      const hover = e.target.closest('a, button, [data-cursor-hover]');
      root.classList.toggle('is-hover', !!hover);
    });
    document.addEventListener('pointerleave', () => root.classList.remove('is-visible'));
    addEventListener('pointerdown', () => root.classList.add('is-down'));
    addEventListener('pointerup', () => root.classList.remove('is-down'));

    (function loop() {
      rx += (mx - rx) * .18;   // 0.18 ずつ近づく（小さいほどゆっくり）
      ry += (my - ry) * .18;
      ring.style.transform = `translate(${rx}px, ${ry}px)`;
      requestAnimationFrame(loop);
    })();
  });
})();

// 図形ごとに「元の位置」と「ずれ」を持ち、マウスから押し出す力とばねで戻る力を毎フレーム計算
(() => {
  const colors = ['#ff3d7f', '#ffd400', '#00c2ff', '#7cff4f', '#ff7a00', '#a259ff'];
  const kinds = ['is-circle', 'is-ring', 'is-tri', 'is-star', 'is-square'];
  document.querySelectorAll('[data-fx="repel"]').forEach((root) => {
    const count = +root.dataset.count || 24;
    const R = 140;                         // マウスの影響が届く半径
    let mouse = { x: -9999, y: -9999 };
    const items = Array.from({ length: count }, (_, i) => {
      const el = document.createElement('i');
      el.className = `fx-repel__shape ${kinds[i % kinds.length]}`;
      el.style.setProperty('--s', `${24 + Math.random() * 34}px`);
      el.style.setProperty('--c', colors[i % colors.length]);
      el.setAttribute('aria-hidden', 'true');
      root.appendChild(el);
      return { el, bx: Math.random(), by: Math.random(), ox: 0, oy: 0, vx: 0, vy: 0, rot: Math.random() * 360 };
    });

    root.addEventListener('pointermove', (e) => {
      const r = root.getBoundingClientRect();
      mouse = { x: e.clientX - r.left, y: e.clientY - r.top };
    });
    root.addEventListener('pointerleave', () => { mouse = { x: -9999, y: -9999 }; });

    (function loop() {
      const W = root.clientWidth, H = root.clientHeight;
      items.forEach((p) => {
        const x = p.bx * W + p.ox, y = p.by * H + p.oy;
        const dx = x - mouse.x, dy = y - mouse.y, d = Math.hypot(dx, dy) || 1;
        if (d < R) {                       // 近いほど強く押し出す
          const f = (R - d) / R * 4;
          p.vx += dx / d * f; p.vy += dy / d * f;
        }
        p.vx += -p.ox * .04; p.vy += -p.oy * .04;   // ばねで元の位置へ
        p.vx *= .86; p.vy *= .86;                   // 少しずつ止まる
        p.ox += p.vx; p.oy += p.vy;
        p.el.style.transform = `translate(${x}px, ${y}px) rotate(${p.rot + p.ox}deg)`;
      });
      requestAnimationFrame(loop);
    })();
  });
})();

// クリックした場所から、花火のように粒子が放射状にはじける
(() => {
  const colors = ['#ff3d7f', '#ffd400', '#00c2ff', '#7cff4f', '#a259ff', '#ffffff'];
  document.querySelectorAll('[data-fx="fireworks"]').forEach((cv) => {
    const ctx = cv.getContext('2d');
    let W = 0, H = 0;
    const resize = () => {
      const d = window.devicePixelRatio || 1;
      W = innerWidth; H = innerHeight;
      cv.width = W * d; cv.height = H * d;
      ctx.setTransform(d, 0, 0, d, 0, 0);
    };
    resize();
    addEventListener('resize', resize);

    const parts = [];
    const burst = (x, y) => {
      const n = 28;
      const base = colors[Math.floor(Math.random() * colors.length)];
      for (let i = 0; i < n; i++) {
        const a = (Math.PI * 2 * i) / n + Math.random() * .25;
        const sp = 2 + Math.random() * 4.5;
        parts.push({
          x, y,
          vx: Math.cos(a) * sp,
          vy: Math.sin(a) * sp,
          r: 1.5 + Math.random() * 2.5,
          life: 1,
          c: Math.random() < .55 ? base : colors[Math.floor(Math.random() * colors.length)],
        });
      }
    };
    addEventListener('pointerdown', (e) => burst(e.clientX, e.clientY));

    (function loop() {
      ctx.clearRect(0, 0, W, H);
      for (let i = parts.length - 1; i >= 0; i--) {
        const p = parts[i];
        p.life -= .018;
        p.vy += .06;              // 重力
        p.vx *= .985; p.vy *= .985;
        p.x += p.vx; p.y += p.vy; p.r *= .985;
        if (p.life <= 0) { parts.splice(i, 1); continue; }
        ctx.globalAlpha = Math.max(0, p.life);
        ctx.fillStyle = p.c;
        ctx.beginPath();
        ctx.arc(p.x, p.y, p.r, 0, Math.PI * 2);
        ctx.fill();
      }
      ctx.globalAlpha = 1;
      requestAnimationFrame(loop);
    })();
  });
})();
