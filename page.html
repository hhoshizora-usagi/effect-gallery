// ①しまで隠す → ②中身を差しかえる → ③しまを開ける
// 本番で別ページへ移動するときは、②で location.href = 'next.html' にする
(() => {
  const cover = document.querySelector('.fx-st');
  const word = cover.querySelector('.fx-st__word');
  const wait = (ms) => new Promise((r) => setTimeout(r, ms));
  let busy = false;

  document.querySelectorAll('[data-fx="stripe-transition"]').forEach((btn) => {
    btn.addEventListener('click', async () => {
      if (busy) return;
      busy = true;
      word.textContent = btn.dataset.word || 'GO!!';
      cover.classList.remove('is-out');
      cover.classList.add('is-in');
      await wait(1100);

      // ② 中身の差しかえ（見本では PAGE 1 ⇔ PAGE 2）
      const page = document.querySelector('.fx-st-page');
      const next = page.dataset.page === '1' ? '2' : '1';
      page.dataset.page = next;
      page.querySelector('.fx-st-page__no').textContent = `PAGE ${next}`;

      cover.classList.add('is-out');
      await wait(700);
      cover.classList.remove('is-in', 'is-out');
      busy = false;
    });
  });
})();

// 見本では2秒で 0→100%。本番では画像の読み込み数などに合わせて p を進める
(() => {
  document.querySelectorAll('[data-fx="loader"]').forEach((el) => {
    const num = el.querySelector('.fx-loader__num');
    const dur = 2000, start = performance.now();
    const tick = (now) => {
      const t = Math.min(1, (now - start) / dur);
      const p = Math.round((1 - Math.pow(1 - t, 2)) * 100);
      num.textContent = `${p}%`;
      el.style.setProperty('--p', `${p}%`);
      if (t < 1) return requestAnimationFrame(tick);
      setTimeout(() => el.classList.add('is-done'), 250);
      setTimeout(() => el.classList.add('is-gone'), 1200);
    };
    requestAnimationFrame(tick);
  });
})();

// <dialog> の showModal() で開く（Esc やフォーカス管理はブラウザがやってくれる）
(() => {
  const close = (dlg) => {
    dlg.classList.add('is-closing');
    dlg.addEventListener('animationend', () => {
      dlg.classList.remove('is-closing');
      dlg.close();
    }, { once: true });
  };
  document.querySelectorAll('[data-fx="modal"]').forEach((btn) => {
    const dlg = document.getElementById(btn.dataset.target);
    btn.addEventListener('click', () => dlg.showModal());
    dlg.querySelectorAll('[data-close]').forEach((b) => b.addEventListener('click', () => close(dlg)));
    dlg.addEventListener('click', (e) => { if (e.target === dlg) close(dlg); });   // 外側クリックで閉じる
    dlg.addEventListener('cancel', (e) => { e.preventDefault(); close(dlg); });     // Esc でもアニメ付きで
  });
})();

// pointer イベントでつまんだ位置を覚えて、動かした分だけ left / top を変える
(() => {
  document.querySelectorAll('[data-fx="sticker"]').forEach((area) => {
    let z = 10;
    area.querySelectorAll('.fx-sticker').forEach((st) => {
      st.addEventListener('pointerdown', (e) => {
        st.setPointerCapture(e.pointerId);
        st.classList.add('is-dragging');
        st.style.zIndex = ++z;                       // つまんだものを一番上に
        const a = area.getBoundingClientRect();
        const offX = e.clientX - a.left - st.offsetLeft;   // つまんだ位置のずれ
        const offY = e.clientY - a.top - st.offsetTop;
        const move = (ev) => {
          const x = Math.max(0, Math.min(a.width - st.offsetWidth, ev.clientX - a.left - offX));
          const y = Math.max(0, Math.min(a.height - st.offsetHeight, ev.clientY - a.top - offY));
          st.style.left = `${x}px`;
          st.style.top = `${y}px`;
        };
        const up = () => {
          st.classList.remove('is-dragging');
          st.removeEventListener('pointermove', move);
          st.removeEventListener('pointerup', up);
          st.removeEventListener('pointercancel', up);
        };
        st.addEventListener('pointermove', move);
        st.addEventListener('pointerup', up);
        st.addEventListener('pointercancel', up);
      });
    });
  });
})();
