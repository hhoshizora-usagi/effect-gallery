  place-items: center;
  width: 100%;
  height: 100vh;

}
.fx-flip__card {
  position: relative;
  width: min(300px, 70vw);
  height: min(400px, 52vh);
  transform-style: preserve-3d;
  transition: transform .7s cubic-bezier(.2, .7, .2, 1);
  cursor: pointer;
}
.fx-flip__card:hover, .fx-flip__card.is-open { transform: rotateY(180deg); }
.fx-flip__face {
  position: absolute;
  inset: 0;
  display: grid;
  place-content: center;
  gap: .4em;
  padding: 20px;
  border-radius: 18px;
  backface-visibility: hidden;
  color: #fff;
  text-align: center;
}
.fx-flip__front { background: linear-gradient(135deg, #ff3d7f, #a259ff); }
.fx-flip__back { background: linear-gradient(135deg, #00c2ff, #00e0c6); transform: rotateY(180deg); }
.fx-flip__face h3 { margin: 0; font-size: 30px; letter-spacing: .08em; }
.fx-flip__face p { margin: 0; font-size: 14px; opacity: .9; }
.fx-flip__icon { font-size: 40px; }
// タップ（クリック）でも裏返るようにする
(() => {
  document.querySelectorAll('.fx-flip__card').forEach((c) => {
    c.addEventListener('click', () => c.classList.toggle('is-open'));
  });
})();

06

3Dパララックス

Parallax 3D

  • HTML
  • CSS
  • JS

マウスの動きに合わせて、奥行きの違う層がずれて立体に見える。

使いどころヒーロー・ビジュアル・芸術的な表現

単体で開く ↗ 
<div class="fx-para" data-fx="parallax3d">
  <div class="fx-para__layer fx-para__back" data-depth="0.25">BACK</div>
  <div class="fx-para__layer fx-para__mid" data-depth="0.6">MID</div>
  <div class="fx-para__layer fx-para__front" data-depth="1">FRONT</div>
</div>
.fx-para {
  position: relative;
  display: grid;
  place-items: center;
  width: 100%;
  height: 100vh;
  overflow: hidden;
  background: radial-gradient(circle at 50% 50%, #14203a, #06070d 75%);
}
.fx-para__layer {
  position: absolute;
  font: 700 clamp(36px, 9vw, 84px) "Dela Gothic One", sans-serif;
  letter-spacing: .08em;
  will-change: transform;
  transition: transform .25s ease-out;
}
.fx-para__back  { color: rgba(0, 194, 255, .35); transform: translateZ(0); }
.fx-para__mid   { color: rgba(255, 61, 127, .6); }
.fx-para__front { color: #fff; text-shadow: 0 0 24px #00e0c6; }
// マウスの位置に応じて、層ごとに違う量ずらす
(() => {
  document.querySelectorAll('[data-fx="parallax3d"]').forEach((host) => {
    const layers = host.querySelectorAll('[data-depth]');
    host.addEventListener('pointermove', (e) => {
      const r = host.getBoundingClientRect();
      const x = (e.clientX - r.left) / r.width - .5;
      const y = (e.clientY - r.top) / r.height - .5;
      layers.forEach((l) => {
        const d = parseFloat(l.dataset.depth) || 0.5;
        l.style.transform = `translate(${-x * d * 60}px, ${-y * d * 60}px)`;
      });
    });
    host.addEventListener('pointerleave', () => {
      layers.forEach((l) => { l.style.transform = ''; });
    });
  });
})();
