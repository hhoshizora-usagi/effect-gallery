リストの項目をつまんで上下にドラッグすると、ほかの項目がスッとよけて順番を入れかえられる。キーボードの↑↓でもOK。

// つまんだ項目を指に合わせて動かし、半分以上かさなったら隣とDOMの順番を入れかえる
(() => {
  document.querySelectorAll('[data-fx="drag-sort"]').forEach((list) => {
    list.addEventListener('pointerdown', (e) => {
      const item = e.target.closest('li');
      if (!item) return;
      item.setPointerCapture(e.pointerId);
      item.classList.add('is-dragging');
      let startY = e.clientY;
      const move = (ev) => {
        let dy = ev.clientY - startY;
        const h = item.offsetHeight + 10;            // 項目の高さ＋すきま
        const prev = item.previousElementSibling, next = item.nextElementSibling;
        if (dy > h / 2 && next) { next.after(item); startY += h; dy -= h; }
        if (dy < -h / 2 && prev) { prev.before(item); startY -= h; dy += h; }
        item.style.transform = `translateY(${dy}px) rotate(${dy / 40}deg) scale(1.04)`;
      };
      const up = () => {
        item.classList.remove('is-dragging');
        item.style.transform = '';
        item.removeEventListener('pointermove', move);
        item.removeEventListener('pointerup', up);
        item.removeEventListener('pointercancel', up);
      };
      item.addEventListener('pointermove', move);
      item.addEventListener('pointerup', up);
      item.addEventListener('pointercancel', up);
    });
    // キーボード：↑↓ で1つずつ移動
    list.addEventListener('keydown', (e) => {
      const item = e.target.closest('li');
      if (!item) return;
      if (e.key === 'ArrowUp' && item.previousElementSibling) { item.previousElementSibling.before(item); }
      else if (e.key === 'ArrowDown' && item.nextElementSibling) { item.nextElementSibling.after(item); }
      else return;
      e.preventDefault();
      item.focus();
    });
  });
})();

// 値（0〜100）を --p に入れて、吹き出しの位置・色・顔文字を変える
(() => {
  const faces = ['😭', '😢', '😐', '🙂', '😆', '🤩'];
  document.querySelectorAll('[data-fx="slider"]').forEach((root) => {
    const input = root.querySelector('input');
    const wrap = root.querySelector('.fx-slider__wrap');
    const bubble = root.querySelector('output');

      const p = +input.value;
      wrap.style.setProperty('--p', p);
      bubble.textContent = `${faces[Math.min(faces.length - 1, Math.floor(p / (100 / faces.length)))]} ${p}`;
    };
    input.addEventListener('input', update);
    update();
  });
})();

// 1文字打つごとに、カーソルのあたりから光のつぶを飛ばし、枠の色（色相 --h）を少し進める
(() => {
  document.querySelectorAll('[data-fx="glow-type"]').forEach((root) => {
    const input = root.querySelector('input');
    const measure = document.createElement('span');          // 文字の幅を測るための見えない箱
    measure.style.cssText = 'position:absolute;visibility:hidden;white-space:pre;font:700 20px "Zen Kaku Gothic New",sans-serif';
    root.appendChild(measure);
    let h = 280;
    input.addEventListener('input', () => {
      h = (h + 23) % 360;
      input.style.setProperty('--h', h);
      measure.textContent = input.value.slice(0, input.selectionStart || input.value.length);
      const x = Math.min(input.offsetLeft + 22 + measure.offsetWidth, input.offsetLeft + input.offsetWidth - 20);
      const y = input.offsetTop + input.offsetHeight / 2;
      for (let i = 0; i < 8; i++) {
        const s = document.createElement('i');
        s.className = 'fx-gtype__spark';
        const a = Math.random() * Math.PI * 2, d = 20 + Math.random() * 40;
        s.style.left = `${x}px`;
        s.style.top = `${y}px`;
        s.style.setProperty('--x', `${Math.cos(a) * d}px`);
        s.style.setProperty('--y', `${Math.sin(a) * d}px`);
        s.style.setProperty('--c', `hsl(${(h + i * 30) % 360} 100% 65%)`);
        root.appendChild(s);
        s.addEventListener('animationend', () => s.remove());
      }
      input.classList.remove('is-pulse');
      void input.offsetWidth;
      input.classList.add('is-pulse');
    });
  });
})();

// ホバー中は「その星まで」を光らせ、選んだら選んだ数まで点灯。中身は本物のラジオボタン
(() => {
  const msgs = ['', 'ざんねん…', 'まあまあ', 'いい感じ！', 'すごくいい！', 'さいこう！！🎉'];
  document.querySelectorAll('[data-fx="rating"]').forEach((root) => {
    const labels = [...root.querySelectorAll('label')];
    const inputs = [...root.querySelectorAll('input')];
    const msg = root.querySelector('.fx-rate__msg');
    const paint = (n, cls) => labels.forEach((l, i) => l.classList.toggle(cls, i < n));
    labels.forEach((l, i) => {
      l.addEventListener('mouseenter', () => paint(i + 1, 'is-hover'));
      l.addEventListener('mouseleave', () => paint(0, 'is-hover'));
    });
    inputs.forEach((input, i) => {
      input.addEventListener('change', () => {
        paint(i + 1, 'is-on');
        msg.textContent = `${i + 1}つ星：${msgs[i + 1]}`;
        labels.forEach((l, k) => {
          l.classList.remove('is-pop');
          if (k <= i) { void l.offsetWidth; l.style.animationDelay = `${k * .06}s`; l.classList.add('is-pop'); }
        });
      });
    });
  });
})();
