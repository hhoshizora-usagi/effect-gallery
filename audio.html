// Web Audio API で音楽を「その場で合成」して鳴らし、AnalyserNode で音の強さを読みとって描く
// （音声ファイルを使うときは、new Audio('song.mp3') を createMediaElementSource につなげばOK）
(() => {
  document.querySelectorAll('[data-fx="visualizer"]').forEach((root) => {
    const cv = root.querySelector('canvas');
    const btn = root.querySelector('button');
    const ctx = cv.getContext('2d');
    let ac, analyser, data, timer, playing = false;

    const note = (freq, time, len, type = 'square', vol = .06) => {
      const o = ac.createOscillator(), g = ac.createGain();
      o.type = type; o.frequency.value = freq;
      g.gain.setValueAtTime(vol, time);
      g.gain.exponentialRampToValueAtTime(.0001, time + len);
      o.connect(g).connect(analyser);
      o.start(time); o.stop(time + len);
    };
    const kick = (time) => {
      const o = ac.createOscillator(), g = ac.createGain();
      o.frequency.setValueAtTime(150, time);
      o.frequency.exponentialRampToValueAtTime(40, time + .2);
      g.gain.setValueAtTime(.35, time);
      g.gain.exponentialRampToValueAtTime(.001, time + .25);
      o.connect(g).connect(analyser);
      o.start(time); o.stop(time + .25);
    };
    // 1小節ぶんのフレーズ（ドミソの分散和音＋キック）を、少し先まで予約しつづける
    const melody = [523, 659, 784, 1047, 784, 659, 587, 698];
    let step = 0, nextTime = 0;
    const schedule = () => {
      while (nextTime < ac.currentTime + .2) {
        if (step % 2 === 0) kick(nextTime);
        note(melody[step % melody.length], nextTime, .2);
        if (step % 4 === 0) note(melody[step % melody.length] / 4, nextTime, .45, 'sawtooth', .05);
        nextTime += .18; step++;
      }
    };
    const start = () => {
      ac = ac || new (window.AudioContext || window.webkitAudioContext)();
      if (!analyser) {
        analyser = ac.createAnalyser();
        analyser.fftSize = 128;
        analyser.connect(ac.destination);
        data = new Uint8Array(analyser.frequencyBinCount);
      }
      ac.resume();
      nextTime = ac.currentTime + .05;
      timer = setInterval(schedule, 50);
      playing = true;
    };
    const stop = () => { clearInterval(timer); playing = false; };
    btn.addEventListener('click', () => {
      playing ? stop() : start();
      btn.textContent = playing ? '❚❚' : '▶';
      btn.setAttribute('aria-pressed', String(playing));
      btn.setAttribute('aria-label', playing ? '音楽を止める' : '音楽を再生');
    });

    let W, H;
    const resize = () => {
      const d = window.devicePixelRatio || 1;
      W = cv.clientWidth; H = cv.clientHeight;
      cv.width = W * d; cv.height = H * d;
      ctx.setTransform(d, 0, 0, d, 0, 0);
    };
    resize();
    addEventListener('resize', resize);

    (function loop(t) {
      ctx.clearRect(0, 0, W, H);
      const bars = 48, R = W * .22;
      let sum = 0;
      if (analyser) analyser.getByteFrequencyData(data);
      for (let i = 0; i < bars; i++) {
        // 音がないときは、ゆるく波うつ待機アニメーション
        const v = analyser && playing ? data[i % data.length] / 255 : (Math.sin((t || 0) / 500 + i * .5) + 1) * .08;
        sum += v;
        const a = (i / bars) * Math.PI * 2 - Math.PI / 2;
        const len = 8 + v * W * .25;
        ctx.strokeStyle = `hsl(${(i / bars) * 360} 100% 62%)`;
        ctx.lineWidth = 6;
        ctx.lineCap = 'round';
        ctx.beginPath();
        ctx.moveTo(W / 2 + Math.cos(a) * R, H / 2 + Math.sin(a) * R);
        ctx.lineTo(W / 2 + Math.cos(a) * (R + len), H / 2 + Math.sin(a) * (R + len));
        ctx.stroke();
      }
      btn.style.setProperty('--beat', (1 + (sum / bars) * .25).toFixed(3));   // 音に合わせてボタンもはねる
      requestAnimationFrame(loop);
    })();
  });
})();

// マウスの x で周波数、y で音量を変える。AnalyserNode の「波形データ」をそのまま線で描く
(() => {
  document.querySelectorAll('[data-fx="theremin"]').forEach((pad) => {
    const cv = pad.querySelector('canvas');
    const ctx = cv.getContext('2d');
    let ac, osc, gain, analyser, wave;
    const init = () => {
      ac = new (window.AudioContext || window.webkitAudioContext)();
      osc = ac.createOscillator();
      gain = ac.createGain();
      analyser = ac.createAnalyser();
      analyser.fftSize = 1024;
      wave = new Uint8Array(analyser.fftSize);
      osc.type = 'triangle';
      gain.gain.value = 0;
      osc.connect(gain).connect(analyser).connect(ac.destination);
      osc.start();
    };
    const play = (e) => {
      if (!ac) init();
      ac.resume();
      const r = pad.getBoundingClientRect();
      const x = Math.min(1, Math.max(0, (e.clientX - r.left) / r.width));
      const y = Math.min(1, Math.max(0, (e.clientY - r.top) / r.height));
      const freq = 130 * Math.pow(2, x * 3);                 // 3オクターブぶん
      osc.frequency.setTargetAtTime(freq, ac.currentTime, .03);
      gain.gain.setTargetAtTime((1 - y) * .18, ac.currentTime, .05);
    };
    const silence = () => { if (gain) gain.gain.setTargetAtTime(0, ac.currentTime, .08); };
    pad.addEventListener('pointerdown', play);
    pad.addEventListener('pointermove', play);
    pad.addEventListener('pointerleave', silence);
    pad.addEventListener('pointerup', (e) => { if (e.pointerType === 'touch') silence(); });

    let W, H;
    const resize = () => {
      const d = window.devicePixelRatio || 1;
      W = cv.clientWidth; H = cv.clientHeight;
      cv.width = W * d; cv.height = H * d;
      ctx.setTransform(d, 0, 0, d, 0, 0);
    };
    resize();
    addEventListener('resize', resize);

    (function loop(t) {
      ctx.clearRect(0, 0, W, H);
      ctx.lineWidth = 3;
      ctx.strokeStyle = '#7cff4f';
      ctx.shadowColor = '#7cff4f';
      ctx.shadowBlur = 10;
      ctx.beginPath();
      if (analyser) analyser.getByteTimeDomainData(wave);
      for (let x = 0; x <= W; x += 2) {
        const i = Math.floor((x / W) * (wave ? wave.length : 1));
        const v = wave ? (wave[i] - 128) / 128 : Math.sin(x / 30 + (t || 0) / 400) * .03;
        const y = H / 2 + v * H * .45;
        x === 0 ? ctx.moveTo(x, y) : ctx.lineTo(x, y);
      }
      ctx.stroke();
      ctx.shadowBlur = 0;
      requestAnimationFrame(loop);
    })();
  });
})();

// それぞれの効果音を「音の高さの動き」と「音量の消え方」で作る（音声ファイルなし）
(() => {
  let ac;
  const tone = (type, from, to, len, vol = .2, delay = 0) => {
    const t = ac.currentTime + delay;
    const o = ac.createOscillator(), g = ac.createGain();
    o.type = type;
    o.frequency.setValueAtTime(from, t);
    o.frequency.exponentialRampToValueAtTime(to, t + len);
    g.gain.setValueAtTime(vol, t);
    g.gain.exponentialRampToValueAtTime(.0001, t + len);
    o.connect(g).connect(ac.destination);
    o.start(t); o.stop(t + len);
  };
  const sounds = {
    pop:   () => tone('sine', 900, 200, .12, .3),
    coin:  () => { tone('square', 988, 988, .08, .12); tone('square', 1319, 1319, .35, .12, .08); },
    jump:  () => tone('square', 220, 880, .25, .12),
    laser: () => tone('sawtooth', 1800, 120, .35, .1),
  };
  document.querySelectorAll('[data-fx="sound-buttons"] [data-sound]').forEach((btn) => {
    btn.addEventListener('click', () => {
      ac = ac || new (window.AudioContext || window.webkitAudioContext)();
      ac.resume();
      sounds[btn.dataset.sound]();
      btn.classList.remove('is-play');
      void btn.offsetWidth;
      btn.classList.add('is-play');
    });
  });
})();
