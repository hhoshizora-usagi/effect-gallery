単体で開く ↗ 
<div class="fx-mag-img" data-fx="magnifier" data-zoom="2.5">
  <!-- 画像は自分のものに差しかえてね -->
  <img src="../../_shared/sample-stage.svg" alt="ライブステージのイラスト">
  <span class="fx-mag-img__lens" aria-hidden="true"></span>
</div>
.fx-mag-img {
  position: relative;
  width: min(90vw, 480px);
  border: 4px solid #111;
  border-radius: 18px;
  overflow: hidden;
  cursor: none;
  box-shadow: 8px 8px 0 #111;
  touch-action: none;
}
.fx-mag-img img { display: block; width: 100%; }
.fx-mag-img__lens {
  position: absolute;
  top: 0;
  left: 0;
  width: 150px;
  height: 150px;
  border: 4px solid #fff;
  border-radius: 50%;
  background-repeat: no-repeat;
  box-shadow: 0 0 0 4px #111, 0 10px 30px rgba(0, 0, 0, .5);
  pointer-events: none;
  opacity: 0;
  transform: scale(.4);
  transition: opacity .2s, transform .25s cubic-bezier(.2, 1.6, .4, 1);
}
.fx-mag-img.is-on .fx-mag-img__lens { opacity: 1; transform: scale(1); }
// レンズの背景に同じ画像を拡大して敷き、マウスの位置に合わせてずらす
(() => {
  document.querySelectorAll('[data-fx="magnifier"]').forEach((root) => {
    const img = root.querySelector('img');
    const lens = root.querySelector('.fx-mag-img__lens');
    const zoom = parseFloat(root.dataset.zoom) || 2.5;
    lens.style.backgroundImage = `url("${img.src}")`;

    root.addEventListener('pointermove', (e) => {
      const r = img.getBoundingClientRect();
      const x = e.clientX - r.left, y = e.clientY - r.top;
      const half = lens.offsetWidth / 2;
      lens.style.left = `${x - half}px`;
      lens.style.top = `${y - half}px`;
      lens.style.backgroundSize = `${r.width * zoom}px ${r.height * zoom}px`;
      lens.style.backgroundPosition = `${-(x * zoom - half)}px ${-(y * zoom - half)}px`;
      root.classList.add('is-on');
    });
    root.addEventListener('pointerleave', () => root.classList.remove('is-on'));
  });
})();

  width: 12px;
  height: 12px;
  padding: 0;
  border: 2px solid #111;
  border-radius: 99px;
  background: #fff;
  cursor: pointer;
  transition: width .3s, background .3s;

// --i（何枚目か）を変えて track を横にずらす。自動再生・スワイプつき
(() => {
  document.querySelectorAll('[data-fx="carousel"]').forEach((root) => {
    const track = root.querySelector('.fx-car__track');
    const slides = [...root.querySelectorAll('.fx-car__slide')];
    const dotsBox = root.querySelector('.fx-car__dots');
    const interval = +root.dataset.interval || 4000;
    let i = 0, timer;

    const dots = slides.map((_, n) => {
      const b = document.createElement('button');
      b.setAttribute('aria-label', `${n + 1}枚目へ`);
      b.addEventListener('click', () => go(n));
      dotsBox.appendChild(b);
      return b;
    });
    const go = (n) => {
      i = (n + slides.length) % slides.length;
      track.style.setProperty('--i', i);
      slides.forEach((s, k) => {
        s.classList.toggle('is-current', k === i);
        s.setAttribute('aria-hidden', String(k !== i));
      });
      dots.forEach((d, k) => d.setAttribute('aria-current', String(k === i)));
      restart();
    };
    const restart = () => { clearInterval(timer); timer = setInterval(() => go(i + 1), interval); };

    root.querySelector('.fx-car__arrow--prev').addEventListener('click', () => go(i - 1));
    root.querySelector('.fx-car__arrow--next').addEventListener('click', () => go(i + 1));
    root.addEventListener('mouseenter', () => clearInterval(timer));   // マウスを乗せている間は止める
    root.addEventListener('mouseleave', restart);

    let startX = null;                                                 // スワイプ
    root.addEventListener('pointerdown', (e) => { startX = e.clientX; });
    root.addEventListener('pointerup', (e) => {
      if (startX === null) return;
      const dx = e.clientX - startX;
      if (Math.abs(dx) > 40) go(i + (dx < 0 ? 1 : -1));
      startX = null;
    });
    go(0);
  });
})();

  100% { transform: scaleX(0); transform-origin: right; }

// 画面に入ったら .is-in をつけて、帯 → 画像 の順にアニメーション
(() => {
  const io = new IntersectionObserver((entries) => {
    entries.forEach((en) => {
      if (en.isIntersecting) { en.target.classList.add('is-in'); io.unobserve(en.target); }
    });
  }, { threshold: .4 });
  document.querySelectorAll('[data-fx="wipe-reveal"]').forEach((el) => io.observe(el));
})();
