波打つ文字

Wave Letters

  • HTML
  • CSS
  • JS

1文字ずつ時間差でぴょこぴょこ跳ねて、色もカラフルに変わる見出し。

使いどころトップの大見出し・イベント名・セール告知

単体で開く ↗ 
<h1 class="fx-wave" data-fx="wave">EPIC DAY!!</h1>
.fx-wave {
  margin: 0;
  font-size: clamp(44px, 12vw, 104px);
  line-height: 1;
  letter-spacing: .02em;
}
.fx-wave span {
  display: inline-block;
  color: var(--c);
  text-shadow: .06em .06em 0 #000;
  animation: fx-wave 1.4s ease-in-out infinite;
  animation-delay: calc(var(--i) * .08s);
}
@keyframes fx-wave {
  0%, 100% { transform: translateY(0) rotate(0); }
  25%      { transform: translateY(-.28em) rotate(-8deg); }
  50%      { transform: translateY(0) rotate(0); }
}
// 文字を1文字ずつ <span> に分けて、順番に跳ねさせる
(() => {
  const colors = ['#ff3d7f', '#ffd400', '#00c2ff', '#7cff4f', '#ff7a00', '#a259ff'];
  document.querySelectorAll('[data-fx="wave"]').forEach((el) => {
    const text = el.textContent.trim();
    el.setAttribute('aria-label', text);   // 読み上げ用に元の文字を残す
    el.textContent = '';
    [...text].forEach((ch, i) => {
      const s = document.createElement('span');
      s.textContent = ch === ' ' ? ' ' : ch;
      s.setAttribute('aria-hidden', 'true');
      s.style.setProperty('--i', i);
      s.style.setProperty('--c', colors[i % colors.length]);
      el.appendChild(s);
    });
  });
})();

02

タイピング

Typewriter

  • HTML
  • CSS
  • JS

言葉を1文字ずつ打って、消して、次の言葉へ。点滅するカーソル付き。

使いどころキャッチコピー・サービス紹介・ヒーローの一言

単体で開く ↗ 
<p class="fx-type">
  ページを<br>
  <span class="fx-type__word" data-fx="typing"
        data-words='["ド派手に。","カラフルに。","EPICに。","動かそう。"]'></span><span class="fx-type__caret" aria-hidden="true"></span>
</p>
.fx-type {
  margin: 0;
  font-size: clamp(30px, 8vw, 64px);
  line-height: 1.25;
  text-align: center;
}
.fx-type__word {
  color: #ffd400;
  background: linear-gradient(transparent 70%, #ff3d7f 70%);
}
.fx-type__caret {
  display: inline-block;
  width: .08em;
  height: 1em;
  margin-left: .06em;
  vertical-align: -.1em;
  background: #00c2ff;
  animation: fx-caret .8s steps(1) infinite;
}
@keyframes fx-caret { 50% { opacity: 0; } }
// data-words に並べた言葉を、打つ → 待つ → 消す でくり返す
(() => {
  const wait = (ms) => new Promise((r) => setTimeout(r, ms));
  document.querySelectorAll('[data-fx="typing"]').forEach(async (el) => {
    const words = JSON.parse(el.dataset.words || '[]');
    const typeSpeed = 110, deleteSpeed = 50, hold = 1300;
    for (let n = 0; ; n = (n + 1) % words.length) {
      const chars = [...words[n]];
      for (let i = 1; i <= chars.length; i++) { el.textContent = chars.slice(0, i).join(''); await wait(typeSpeed); }
      await wait(hold);
      for (let i = chars.length; i >= 0; i--) { el.textContent = chars.slice(0, i).join(''); await wait(deleteSpeed); }
      await wait(250);
    }
  });
})();

スクランブル

Text Scramble

  • HTML
  • CSS
  • JS

カタカナや記号がパラパラ入れ替わって、最後に本当の文字が決まる。マウスを乗せるともう一度。

使いどころセクション見出し・ローディング明け・SF/テック系

単体で開く ↗ 
<div class="fx-scramble-wrap">
  <h2 class="fx-scramble" data-fx="scramble">STAGE IS READY</h2>
  <h2 class="fx-scramble fx-scramble--jp" data-fx="scramble">ステージ、開幕。</h2>
</div>
.fx-scramble-wrap { text-align: center; }
.fx-scramble {
  margin: .1em 0;
  font-family: "JetBrains Mono", ui-monospace, monospace;
  font-size: clamp(26px, 7vw, 56px);
  font-weight: 800;
  color: #7cff4f;
  cursor: default;
}
.fx-scramble--jp { color: #ff3d7f; font-family: inherit; }
.fx-scramble .is-random { color: #00c2ff; opacity: .8; }
// 左から順に、ランダムな文字 → 本当の文字 に確定させる
(() => {
  const pool = 'アイウエオカキクケコサシスセソタチツテトABCDEFGHIJKLMNOPQRSTUVWXYZ#%&*+?!';
  const rand = () => pool[Math.floor(Math.random() * pool.length)];
  function run(el) {
    const final = [...el.dataset.text];
    let frame = 0;
    cancelAnimationFrame(el._raf);
    const tick = () => {
      el.innerHTML = '';
      final.forEach((ch, i) => {
        const done = frame > i * 3 + 12;          // 3フレームずつずらして確定
        const span = document.createElement('span');
        span.textContent = done || ch === ' ' ? ch : rand();
        if (!done && ch !== ' ') span.className = 'is-random';
        el.appendChild(span);
      });
      frame++;
      if (frame <= final.length * 3 + 12) el._raf = requestAnimationFrame(tick);
    };
    tick();
  }
  document.querySelectorAll('[data-fx="scramble"]').forEach((el) => {
    el.dataset.text = el.textContent.trim();
    el.setAttribute('aria-label', el.dataset.text);
    run(el);
    el.addEventListener('mouseenter', () => run(el));
  });
})();

蛍光マーカー

Highlighter

  • HTML
  • CSS
  • JS

文章が画面に入ると、大事なところに蛍光ペンでシャーッと線を引いたように色がつく。

使いどころブログ・説明文・料金プランの強調ポイント

単体で開く ↗ 
<p class="fx-marker-text">
  エフェクトをひとつ足すだけで、
  <mark class="fx-marker" data-fx="marker">ページの印象はガラッと変わる</mark>。
  まずは<mark class="fx-marker fx-marker--pink" data-fx="marker" style="--d: .9s">コピーして試してみよう！</mark>
</p>
.fx-marker-text {
  max-width: 520px;
  margin: 0;
  padding: 0 20px;
  color: #111;
  font: 700 clamp(18px, 4.4vw, 26px)/1.9 "Zen Kaku Gothic New", system-ui, sans-serif;
}
.fx-marker {
  padding: 0 .1em;
  color: inherit;
  background: linear-gradient(transparent 55%, #ffd400 55%, #ffd400 92%, transparent 92%) no-repeat;
  background-size: 0% 100%;                 /* 最初は幅 0 */
  transition: background-size 1s cubic-bezier(.6, 0, .2, 1) var(--d, .2s);
  -webkit-box-decoration-break: clone;
          box-decoration-break: clone;      /* 2行にまたがってもきれいに */
}
.fx-marker--pink { background-image: linear-gradient(transparent 55%, #ff9ecf 55%, #ff9ecf 92%, transparent 92%); }
.fx-marker.is-in { background-size: 100% 100%; }
// 画面に入ったら .is-in をつけて、マーカーを左から右へ引く
(() => {
  const io = new IntersectionObserver((entries) => {
    entries.forEach((en) => {
      if (en.isIntersecting) { en.target.classList.add('is-in'); io.unobserve(en.target); }
    });
  }, { threshold: .8 });
  document.querySelectorAll('[data-fx="marker"]').forEach((el) => io.observe(el));
})();
