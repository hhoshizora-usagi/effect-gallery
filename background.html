01

紙吹雪の雨

Confetti Rain

  • HTML
  • CSS
  • JS

カラフルな紙吹雪が、くるくる回りながら上からずっと降り続ける背景。

使いどころお祝い・当選発表・イベント告知・誕生日

単体で開く ↗ 
<section class="fx-rain">
  <canvas class="fx-rain__canvas" data-fx="confetti-rain" data-count="120" aria-hidden="true"></canvas>
  <h2 class="fx-rain__title">CONGRATS!</h2>
</section>
.fx-rain {
  position: relative;
  display: grid;
  place-items: center;
  width: 100%;
  height: 100vh;
  overflow: hidden;
}
.fx-rain__canvas {
  position: absolute;
  inset: 0;
  width: 100%;
  height: 100%;
}
.fx-rain__title {
  position: relative;
  margin: 0;
  font-size: clamp(44px, 12vw, 100px);
  color: #fff;
  text-shadow: 6px 6px 0 #a259ff;
}
// 紙を1枚ずつ「位置・速さ・回転」で管理して、下まで落ちたら上に戻す
(() => {
  const colors = ['#ff3d7f', '#ffd400', '#00c2ff', '#7cff4f', '#ff7a00', '#a259ff', '#ffffff'];
  document.querySelectorAll('[data-fx="confetti-rain"]').forEach((cv) => {
    const ctx = cv.getContext('2d');
    let W = 0, H = 0;
    const resize = () => {
      const d = window.devicePixelRatio || 1;
      W = cv.clientWidth; H = cv.clientHeight;
      cv.width = W * d; cv.height = H * d;
      ctx.setTransform(d, 0, 0, d, 0, 0);
    };
    resize();
    addEventListener('resize', resize);

    const make = (anyY) => ({
      x: Math.random() * W,
      y: anyY ? Math.random() * H : -20,
      w: 6 + Math.random() * 8,
      h: 10 + Math.random() * 10,
      vy: 1 + Math.random() * 2.2,
      sway: Math.random() * Math.PI * 2,
      rot: Math.random() * Math.PI,
      vr: (Math.random() - .5) * .2,
      c: colors[Math.floor(Math.random() * colors.length)],
    });
    const bits = Array.from({ length: +cv.dataset.count || 100 }, () => make(true));

    (function loop() {
      ctx.clearRect(0, 0, W, H);
      bits.forEach((b, i) => {
        b.y += b.vy; b.sway += .03; b.rot += b.vr;
        b.x += Math.sin(b.sway) * .8;
        if (b.y > H + 20) bits[i] = make(false);
        ctx.save();
        ctx.translate(b.x, b.y);
        ctx.rotate(b.rot);
        ctx.scale(1, Math.cos(b.sway * 2));   // 裏返るようにひらひら
        ctx.fillStyle = b.c;
        ctx.fillRect(-b.w / 2, -b.h / 2, b.w, b.h);
        ctx.restore();
      });
      requestAnimationFrame(loop);
    })();
  });
})();

// 図形を作って、大きさ・色・動く距離・速さをランダムに決める（動き自体は CSS）
(() => {
  const colors = ['#ff3d7f', '#ffd400', '#00c2ff', '#7cff4f', '#ff7a00', '#a259ff'];
  const kinds = ['is-circle', 'is-ring', 'is-tri', 'is-star', 'is-pill'];
  const rnd = (a, b) => a + Math.random() * (b - a);
  document.querySelectorAll('[data-fx="floating-shapes"]').forEach((root) => {
    const count = +root.dataset.count || 16;
    for (let i = 0; i < count; i++) {
      const el = document.createElement('i');
      el.className = `fx-float__shape ${kinds[i % kinds.length]}`;
      el.setAttribute('aria-hidden', 'true');
      Object.entries({
        '--s': `${rnd(26, 80)}px`,
        '--c': colors[i % colors.length],
        '--d': `${rnd(3, 7)}s`,
        '--delay': `${-rnd(0, 6)}s`,
        '--dx': `${rnd(-60, 60)}px`,
        '--dy': `${rnd(-70, 70)}px`,
        '--r': `${rnd(-200, 200)}deg`,
      }).forEach(([k, v]) => el.style.setProperty(k, v));
      el.style.left = `${rnd(0, 95)}%`;
      el.style.top = `${rnd(0, 92)}%`;
      root.appendChild(el);
    }
  });
})();

つながる星

Particle Network

  • HTML
  • CSS
  • JS

点がふわふわ漂って、近い点どうしが線でつながる。マウスにも線が伸びてくる。

使いどころテック・IT・コミュニティ系のヒーロー

単体で開く ↗ 
<section class="fx-net">
  <canvas class="fx-net__canvas" data-fx="particle-network" data-count="70" aria-hidden="true"></canvas>
  <h2 class="fx-net__title">CONNECT</h2>
</section>
.fx-net {
  position: relative;
  display: grid;
  place-items: center;
  width: 100%;
  height: 100vh;
  overflow: hidden;
  background: radial-gradient(circle at 50% 40%, #2a1d5c, #0b0816 70%);
}
.fx-net__canvas { position: absolute; inset: 0; width: 100%; height: 100%; }
.fx-net__title {
  position: relative;
  margin: 0;
  font-size: clamp(42px, 11vw, 96px);
  letter-spacing: .08em;
  color: #fff;
  pointer-events: none;
  text-shadow: 0 0 24px #00c2ff;
}
// 点を動かして、距離が近いペアだけ線を引く（遠いほど薄く）
(() => {
  const colors = ['#ff3d7f', '#ffd400', '#00c2ff', '#7cff4f', '#a259ff'];
  document.querySelectorAll('[data-fx="particle-network"]').forEach((cv) => {
    const ctx = cv.getContext('2d');
    const LINK = 120;                        // この距離より近いとつながる
    let W = 0, H = 0, mouse = null;
    const resize = () => {
      const d = window.devicePixelRatio || 1;
      W = cv.clientWidth; H = cv.clientHeight;
      cv.width = W * d; cv.height = H * d;
      ctx.setTransform(d, 0, 0, d, 0, 0);
    };
    resize();
    addEventListener('resize', resize);
    const pts = Array.from({ length: +cv.dataset.count || 60 }, (_, i) => ({
      x: Math.random() * W, y: Math.random() * H,
      vx: (Math.random() - .5) * .8, vy: (Math.random() - .5) * .8,
      c: colors[i % colors.length],
    }));
    const host = cv.parentElement;
    host.addEventListener('pointermove', (e) => {
      const r = cv.getBoundingClientRect();
      mouse = { x: e.clientX - r.left, y: e.clientY - r.top };
    });
    host.addEventListener('pointerleave', () => { mouse = null; });

    const line = (a, b, max, color) => {
      const d = Math.hypot(a.x - b.x, a.y - b.y);
      if (d > max) return;
      ctx.globalAlpha = 1 - d / max;
      ctx.strokeStyle = color;
      ctx.beginPath(); ctx.moveTo(a.x, a.y); ctx.lineTo(b.x, b.y); ctx.stroke();
    };

    (function loop() {
      ctx.clearRect(0, 0, W, H);
      ctx.lineWidth = 1;
      pts.forEach((p) => {
        p.x += p.vx; p.y += p.vy;
        if (p.x < 0 || p.x > W) p.vx *= -1;
        if (p.y < 0 || p.y > H) p.vy *= -1;
      });
      for (let i = 0; i < pts.length; i++) {
        for (let j = i + 1; j < pts.length; j++) line(pts[i], pts[j], LINK, '#8fa8ff');
        if (mouse) line(pts[i], mouse, LINK * 1.6, pts[i].c);
      }
      ctx.globalAlpha = 1;
      pts.forEach((p) => {
        ctx.fillStyle = p.c;
        ctx.beginPath(); ctx.arc(p.x, p.y, 2.6, 0, Math.PI * 2); ctx.fill();
      });
      requestAnimationFrame(loop);
    })();
  });
})();

  position: relative;
  display: grid;
  place-items: center;
  width: 100%;
  height: 100vh;
  overflow: hidden;
  background: #111;
}
.fx-dots__canvas { position: absolute; inset: 0; width: 100%; height: 100%; }
.fx-dots__title {
  position: relative;
  margin: 0;
  padding: .15em .4em;
  background: #111;
  color: #fff;
  font-size: clamp(42px, 11vw, 92px);
  line-height: 1;
  text-align: center;
}
// 格子状の点を、中心からの距離と時間で sin 波にして大きさと色を変える
(() => {
  document.querySelectorAll('[data-fx="dot-wave"]').forEach((cv) => {
    const ctx = cv.getContext('2d');
    const GAP = 26;                          // 点の間隔
    let W = 0, H = 0;
    const resize = () => {
      const d = window.devicePixelRatio || 1;
      W = cv.clientWidth; H = cv.clientHeight;
      cv.width = W * d; cv.height = H * d;
      ctx.setTransform(d, 0, 0, d, 0, 0);
    };
    resize();
    addEventListener('resize', resize);

    (function loop(t) {
      t = (t || 0) / 1000;
      ctx.clearRect(0, 0, W, H);
      for (let x = GAP / 2; x < W; x += GAP) {
        for (let y = GAP / 2; y < H; y += GAP) {
          const d = Math.hypot(x - W / 2, y - H / 2) / 60;
          const wave = Math.sin(d - t * 3);          // -1〜1
          const r = 2 + (wave + 1) * 4.5;
          ctx.fillStyle = `hsl(${(d * 30 + t * 60) % 360} 95% 60%)`;
          ctx.beginPath(); ctx.arc(x, y, r, 0, Math.PI * 2); ctx.fill();
        }
      }
      requestAnimationFrame(loop);
    })();
  });
})();
